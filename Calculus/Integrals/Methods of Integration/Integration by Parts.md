
> [!info] Derivation of Integration by Parts
> $$
\begin{align}
\frac{\mathrm{d}}{\mathrm{d} x} f\left( x \right) \cdot g\left( x \right) &=
 f'\left( x \right) g\left( x \right) + f\left( x\right) g'\left( x \right) \\
\int \left[ f'(x) g\left( x \right) + f\left( x \right) g'\left( x \right)    \right] \cdot \mathrm{d} x &= f\left( x \right) g\left( x \right) +C\ \\
\int \left[ f'\left( x \right) g\left( x \right) \right] + \int \left[ f\left( x\right) g'\left( x \right)   \right] &= \\
\int \left[ f'\left( x \right) g\left( x \right) \right] &= f\left( x\right)  g\left( x \right) -\int \left[ f\left( x\right) g'\left( x \right)   \right]
\end{align}
$$
